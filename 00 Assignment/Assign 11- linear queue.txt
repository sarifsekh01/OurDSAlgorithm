Assignment 11
Write a C program to implement a linear queue using an array and perform basic operations using a menu-driven approach.

<>code 

#include <stdio.h>

#define MAX 5

int queue[MAX];
int front = -1;
int rear = -1;

// Enqueue
void enqueue()
{
    int value;

    if (rear == MAX - 1)
    {
        printf("\nQueue Overflow!\n");
    }
    else
    {
        printf("Enter value: ");
        scanf("%d", &value);

        if (front == -1)
            front = 0;

        rear++;
        queue[rear] = value;

        printf("%d inserted successfully.\n", value);
    }
}

// Dequeue
void dequeue()
{
    if (front == -1 || front > rear)
    {
        printf("\nQueue Underflow!\n");
    }
    else
    {
        printf("%d deleted successfully.\n", queue[front]);

        front++;

        if (front > rear)
        {
            front = -1;
            rear = -1;
        }
    }
}

// Display
void display()
{
    int i;

    if (front == -1)
    {
        printf("\nQueue is empty!\n");
    }
    else
    {
        printf("\nQueue elements: ");

        for (i = front; i <= rear; i++)
        {
            printf("%d ", queue[i]);
        }

        printf("\n");
    }
}

int main()
{
    int choice;

    while (1)
    {
        printf("\n===== LINEAR QUEUE =====\n");
        printf("1. Enqueue\n");
        printf("2. Dequeue\n");
        printf("3. Display\n");
        printf("4. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                enqueue();
                break;

            case 2:
                dequeue();
                break;

            case 3:
                display();
                break;

            case 4:
                return 0;

            default:
                printf("Invalid Choice!\n");
        }
    }

    return 0;
}
